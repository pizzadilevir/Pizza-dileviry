        let menuData = JSON.parse(localStorage.getItem('pizza_delivery_menu')) || defaultMenuData;
        let cart = [];
        let currentCategory = 'pizza';

        function setLanguage(lang) {
            document.getElementById('lang-modal').style.display = 'none';
        }

        function renderMenu(category) {
            currentCategory = category;
            const container = document.getElementById('menu-container');
            container.innerHTML = '';

            const filtered = menuData.filter(item => item.cat === category);
            
            let html = `<div class="items-grid">`;
            filtered.forEach(item => {
                html += `
                    <div class="food-card">
                        <div class="food-img-container">
                            <img src="${item.img}" alt="${item.name}" onerror="this.src='https://images.unsplash.com/photo-1513104890138-7c749659a591?w=500'">
                        </div>
                        <div class="food-info">
                            <h4>${item.name}</h4>
                            <div class="food-price">${item.price}</div>
                        </div>
                        <div class="food-actions">
                            <button class="add-to-cart-btn" onclick="addToCart('${item.id}')"><i class="fas fa-plus"></i> اطلب</button>
                            <button class="admin-edit-btn" onclick="openAdminEdit('${item.id}')" style="display:none;" id="edit-btn-${item.id}">تعديل</button>
                        </div>
                    </div>
                `;
            });
            html += `</div>`;
            container.innerHTML = html;
            checkAdminVisibility();
        }

        function filterCategory(cat) {
            document.querySelectorAll('.cat-btn').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            renderMenu(cat);
        }

        function addToCart(id) {
            const item = menuData.find(i => i.id === id);
            const existing = cart.find(i => i.id === id);
            if (existing) {
                existing.qty++;
            } else {
                cart.push({ ...item, qty: 1 });
            }
            updateCartButton();
        }

        function updateCartButton() {
            const totalQty = cart.reduce((sum, i) => sum + i.qty, 0);
            document.getElementById('cart-btn-text').innerText = `السلة (${totalQty})`;
        }

        function openCartModal() {
            const list = document.getElementById('cart-items-list');
            list.innerHTML = '';
            if (cart.length === 0) {
                list.innerHTML = '<p style="text-align:center; color:#888; padding:20px;">السلة فارغة حالياً</p>';
                document.getElementById('cart-total-price').innerText = '0 دينار';
                document.getElementById('cart-modal').style.display = 'flex';
                return;
            }

            let totalPrice = 0;
            cart.forEach((item, index) => {
                list.innerHTML += `
                    <div class="cart-item">
                        <div>
                            <strong>${item.name}</strong><br>
                            <small style="color:#aaa;">السعر: ${item.price}</small>
                        </div>
                        <div style="display:flex; align-items:center; gap:10px;">
                            <span>الكمية: ${item.qty}</span>
                            <button onclick="removeFromCart(${index})" style="background:#e50914; color:#fff; border:none; padding:4px 8px; border-radius:4px; cursor:pointer;"><i class="fas fa-trash"></i></button>
                        </div>
                    </div>
                `;
            });
            document.getElementById('cart-total-price').innerText = 'حسب الطلب في الواتساب';
            document.getElementById('cart-modal').style.display = 'flex';
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartButton();
            openCartModal();
        }

        function closeCartModal() {
            document.getElementById('cart-modal').style.display = 'none';
        }

        function sendWhatsAppOrder() {
            const name = document.getElementById('cust-name').value;
            const phone = document.getElementById('cust-phone').value;
            const address = document.getElementById('cust-address').value;

            if (!name || !phone) {
                alert('الرجاء إدخال الاسم ورقم الهاتف على الأقل');
                return;
            }

            let message = `*طلب جديد من موقع بيتزا دليفري*%0A`;
            message += `👤 *الاسم:* ${name}%0A`;
            message += `📞 *الهاتف:* ${phone}%0A`;
            message += `📍 *العنوان:* ${address}%0A%0A`;
            message += `🛒 *الطلبات:*%0A`;
            cart.forEach(i => {
                message += `- ${i.name} (العدد: ${i.qty}) - السعر: ${i.price}%0A`;
            });

            const whatsappUrl = `https://wa.me/9647508653006?text=${message}`;
            window.open(whatsappUrl, '_blank');
        }

        // نظام الإدارة والتعديل بالكود 30067
        let isAdminLoggedIn = false;

        function openAdminLogin() {
            const code = prompt('أدخل كود الدخول إلى قائمة التعديل:');
            if (code === '30067') {
                isAdminLoggedIn = true;
                alert('تم تسجيل الدخول بنجاح كمسؤول! أزرار التعديل ظاهرة الآن.');
                checkAdminVisibility();
            } else if (code !== null) {
                alert('الكود غير صحيح!');
            }
        }

        function checkAdminVisibility() {
            document.querySelectorAll('.admin-edit-btn').forEach(btn => {
                btn.style.display = isAdminLoggedIn ? 'block' : 'none';
            });
        }

        function openAdminEdit(id) {
            const item = menuData.find(i => i.id === id);
            document.getElementById('edit-item-id').value = item.id;
            document.getElementById('edit-item-name').value = item.name;
            document.getElementById('edit-item-price').value = item.price;
            document.getElementById('edit-item-img').value = item.img;
            document.getElementById('admin-modal').style.display = 'flex';
        }

        function closeAdminModal() {
            document.getElementById('admin-modal').style.display = 'none';
        }

        function saveItemEdit() {
            const id = document.getElementById('edit-item-id').value;
            const item = menuData.find(i => i.id === id);
            item.name = document.getElementById('edit-item-name').value;
            item.price = document.getElementById('edit-item-price').value;
            item.img = document.getElementById('edit-item-img').value;

            // حفظ في التخزين المحلي لضمان بقائها دائماً
            localStorage.setItem('pizza_delivery_menu', JSON.stringify(menuData));
            
            closeAdminModal();
            renderMenu(currentCategory);
            alert('تم حفظ التعديلات بنجاح ولن تضيع أبدًا!');
        }
