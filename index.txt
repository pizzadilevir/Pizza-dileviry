        function renderMenu(category = 'all') {
            const grid = document.getElementById('menu-grid');
            grid.innerHTML = '';
            
            let filtered = category === 'all' ? menuItems : menuItems.filter(i => i.category === category);
            
            filtered.forEach(item => {
                grid.innerHTML += `
                    <div class="menu-card">
                        <img src="${item.img}" class="menu-img" alt="${item.name}">
                        <div class="menu-body">

                            </div>

                    </div>
                `;
            });

        function filterMenu(cat) {
            document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
            event.target.classList.add('active');
            renderMenu(cat);
        }

        function addToCart(name, price) {
            cart.push({ name, price });
            updateCartUI();
        }

        function updateCartUI() {
            const list = document.getElementById('cart-items');
            list.innerHTML = '';
            
            if(cart.length === 0) {
                list.innerHTML = `<li style="color: var(--text-light); text-align: center;">${translations[currentLang].emptyCart}</li>`;
                document.getElementById('total-text').innerText = `${translations[currentLang].total}: 0 د.ع`;
                return;
            }

            let total = 0;
            cart.forEach((item, index) => {
                total += item.price;
                list.innerHTML += `
                    <li class="cart-item">
                        <span>${item.name}</span>
                        <span>${item.price} د.ع</span>
                    </li>
                `;
            });

            document.getElementById('total-text').innerText = `${translations[currentLang].total}: ${total} د.ع`;
        }

        function sendWhatsAppOrder() {
            if(cart.length === 0) {
                alert('السلة فارغة!');
                return;
            }
            let text = "طلب جديد من مطعم زاخو:\n";
            let total = 0;
            cart.forEach(i => {
                text += `- ${i.name} (${i.price} د.ع)\n`;
                total += i.price;
            });
            text += `\nالمجموع الكلي: ${total} د.ع`;
            let url = `https://wa.me/9647500000000?text=${encodeURIComponent(text)}`;
            window.open(url, '_blank');
        }

        function checkAdmin() {
            let pass = document.getElementById('admin-pass').value;
            if(pass === '30067') {
                document.getElementById('admin-login-box').style.display = 'none';
                document.getElementById('admin-panel').style.display = 'block';
            } else {
                alert('كلمة المرور غير صحيحة!');
            }
        }

        function addNewItem() {
            let name = document.getElementById('new-name').value;
            let desc = document.getElementById('new-desc').value;
            let price = Number(document.getElementById('new-price').value);
            let category = document.getElementById('new-cat').value;
            let img = document.getElementById('new-img').value || 'https://images.unsplash.com/photo-1513104890138-7c749659a591?auto=format&fit=crop&w=500&q=80';

            if(!name || !price) {
                alert('يرجى إدخال اسم الوجبة والسعر على الأقل!');
                return;
            }

            menuItems.push({ id: Date.now(), name, desc, price, category, img });
            localStorage.setItem('zakho_menu', JSON.stringify(menuItems));
            renderMenu();
            alert('تم إضافة الوجبة بنجاح!');
            document.getElementById('new-name').value = '';
            document.getElementById('new-desc').value = '';
            document.getElementById('new-price').value = '';
            document.getElementById('new-img').value = '';
        }
